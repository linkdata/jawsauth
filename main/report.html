const authRefreshSkew = time.Minute
const authSessionCheckInterval = time.Minute

var errOIDCStaleIDToken = errors.New("oidc stale id_token")
var errAuthTimerStale = errors.New("auth timer stale")
var errOIDCInvalidExpiry = errors.New("oidc invalid exp")
var errOIDCMissingSubject = errors.New("oidc missing sub")
var errOIDCSubjectMismatch = errors.New("oidc sub changed on refresh")

type authTimer interface {
        Stop() bool
}

type authTimerAfterFunc func(time.Duration, func()) authTimer

type authTimerState struct {
        timer  authTimer
        expiry time.Time
}

// pendingLoginEvent orders callbacks when authentication is cleared before login notification finishes.
type pendingLoginEvent struct {
        cleared bool
}

func authTimerEntryExpiry(entry *authTimerState) (expiry time.Time) {
        if entry != nil {
                expiry = entry.expiry
        }
        return
}

// debugSessionPrefix returns a short cookie prefix for log correlation.
func debugSessionPrefix(sess *jaws.Session) string {
        value := sess.CookieValue()
        // Keep at least eight base-32 characters out of logs.
        if len(value) >= 12 {
                return value[:4]
        }
        return ""
}

func tokenDebugAttrs(token *oauth2.Token) []any {
        attrs := []any{"token_nil", token == nil}
        if token != nil {
                rawIDToken, _ := token.Extra("id_token").(string)
                attrs = append(attrs,
                        "access_token_present", token.AccessToken != "",
                        "refresh_token_present", token.RefreshToken != "",
                        "token_expiry", token.Expiry,
                        "token_valid", token.Valid(),
                        "id_token_present", rawIDToken != "",
                        "id_token_len", len(rawIDToken),
                )
        }
        return attrs
}

func realAuthTimerAfterFunc(delay time.Duration, callback func()) authTimer {
        return time.AfterFunc(delay, callback)
}

func (srv *Server) oauth2Context(ctx context.Context) (authctx context.Context) {
        authctx = ctx
        if srv != nil {
                client, _ := authctx.Value(oauth2.HTTPClient).(*http.Client)
                if client == nil {
                        client = srv.httpClient
                }
                clientCopy := boundedProviderClient(client)
                // Token POSTs carry secrets, and 307/308 redirects preserve their body.
                clientCopy.CheckRedirect = func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }
                if logger := srv.debugLogger(); logger != nil {
                        next := clientCopy.Transport
                        if next == nil {
                                next = http.DefaultTransport
                        }
                        clientCopy.Transport = debugOAuth2Transport{logger: logger, next: next}
                }
                authctx = context.WithValue(authctx, oauth2.HTTPClient, &clientCopy)
        }
        return
}

func (srv *Server) sessionAuthStatus(sess *jaws.Session, now func() time.Time) (current, present bool) {
        if srv != nil && sess != nil {
                expiryValue := sess.Get(srv.sessionKey(oauth2AuthExpiryKey))
                authValue := sess.Get(srv.SessionKey)
                present = authValue != nil || expiryValue != nil
                if authValue != nil && now != nil {
                        if srv.issuer != "" {
                                claims, ok := authValue.(map[string]any)
                                issuer, _ := claims["iss"].(string)
                                // go-oidc accepts Google's scheme-less issuer.
                                if !ok || (issuer != srv.issuer && (srv.issuer != "https://accounts.google.com" || issuer != "accounts.google.com")) {
                                        return
                                }
                        }
                        if expiry, _ := expiryValue.(time.Time); !expiry.IsZero() {
                                current = expiry.After(now())
                        }
                }
        }
        return
}

func (srv *Server) sessionAccessStatus(sess *jaws.Session) (current, present, pending bool) {
        srv.mu.Lock()
        current, present = srv.sessionAuthStatus(sess, time.Now)
        pending = !srv.RotateSessionOnLogin && srv.loginEvents[sess] != nil
        srv.mu.Unlock()
        return
}

func sessionAlive(sess *jaws.Session) bool {
        cookie := sess.Cookie()
        return cookie != nil && cookie.MaxAge >= 0
}

func (srv *Server) identityExpiryError(expiry time.Time) error {
        if expiry.After(time.Now()) {
                return nil
        }
        if srv.profileSubject != nil {
                return ErrOAuth2InvalidProfile
        }
        return errOIDC{kind: ErrOIDCInvalidIDToken, cause: errOIDCInvalidExpiry}
}

func (srv *Server) verifyIDToken(ctx context.Context, raw string) (idToken *oidc.IDToken, claims map[string]any, err error) {
        if raw == "" {
                return nil, nil, ErrOIDCMissingIDToken
        }
        idToken, err = srv.idTokenVerifier.Verify(ctx, raw)
        if wrapOIDC(ErrOIDCInvalidIDToken, &err) != nil {
                return
        }
        if idToken.Subject == "" {
                return nil, nil, errOIDC{kind: ErrOIDCInvalidIDToken, cause: errOIDCMissingSubject}
        }
        err = idToken.Claims(&claims)
        _ = wrapOIDC(ErrOIDCInvalidIDToken, &err)
        return
}

func (srv *Server) storeSessionAuthClaims(sess *jaws.Session, claims map[string]any, tokenSource oauth2.TokenSource, expiry time.Time, entry *authTimerState) error {
        return srv.storeSessionAuthClaimsIfActive(sess, claims, tokenSource, expiry, entry, nil)
}

func (srv *Server) storeSessionAuthClaimsIfActive(sess *jaws.Session, claims map[string]any, tokenSource oauth2.TokenSource, expiry time.Time, entry *authTimerState, loginCanceled <-chan struct{}) error {
        if srv == nil {
                return ErrOAuth2NotConfigured
        }
        if sess == nil {
                return ErrOAuth2MissingSession
        }
        if expiry.IsZero() {
                return errOIDC{kind: ErrOIDCInvalidIDToken, cause: errOIDCInvalidExpiry}
        }
        email, verified := srv.extractEmail(claims) // Logger callbacks must run outside srv.mu.
        claims["email_verified"] = verified
        delay, replaced, err := srv.storePreparedSessionAuthClaimsIfActive(sess, claims, tokenSource, expiry, entry, loginCanceled, email, verified)
        if err == nil {
                srv.notifyStoredSessionAuthClaims(sess, expiry, delay, replaced)
        }
        return err
}

func (srv *Server) storePreparedSessionAuthClaimsIfActive(sess *jaws.Session, claims map[string]any, tokenSource oauth2.TokenSource, expiry time.Time, entry *authTimerState, loginCanceled <-chan struct{}, email any, verified bool) (delay time.Duration, replaced bool, err error) {
        srv.mu.Lock()
        select {
        case <-loginCanceled:
                srv.mu.Unlock()
                err = ErrOAuth2MissingSession
                return
        default:
        }
        if err = srv.identityExpiryError(expiry); err != nil {
                srv.mu.Unlock()
                return
        }
        if entry != nil && srv.authTimers[sess.ID()] != entry {
                srv.mu.Unlock()
                err = errAuthTimerStale
                return
        }
        if !sessionAlive(sess) {
                srv.mu.Unlock()
                if entry == nil {
                        err = ErrOAuth2MissingSession
                        return
                }
                srv.clearSessionAuth(sess, nil, entry, true)
                err = errAuthTimerStale
                return
        }
        if entry != nil {
                previous, _ := sess.Get(srv.SessionKey).(map[string]any)
                oldSubject, _ := previous["sub"].(string)
                newSubject, _ := claims["sub"].(string)
                if oldSubject == "" || oldSubject != newSubject {
                        srv.mu.Unlock()
                        err = errOIDC{kind: ErrOIDCInvalidIDToken, cause: errOIDCSubjectMismatch}
                        return
                }
        }
        sess.Set(srv.SessionKey, claims)
        sess.Set(srv.SessionTokenKey, tokenSource)
        sess.Set(srv.sessionKey(oauth2AuthExpiryKey), expiry)
        sess.Set(srv.SessionEmailKey, email)
        sess.Set(srv.SessionEmailVerifiedKey, verified)
        delay, replaced = srv.scheduleSessionAuthTimerLocked(sess, expiry)
        if loginCanceled != nil {
                // Keep a new flow from replacing this login before its callback finishes.
                if srv.loginEvents == nil {
                        srv.loginEvents = make(map[*jaws.Session]*pendingLoginEvent)
                }
                srv.loginEvents[sess] = &pendingLoginEvent{}
        }
        srv.mu.Unlock()
        return
}

func (srv *Server) notifyStoredSessionAuthClaims(sess *jaws.Session, expiry time.Time, delay time.Duration, replaced bool) {
        srv.Jaws.Dirty(sess)
        mode := "oidc"
        if srv.profileSubject != nil {
                mode = "profile"
        }
        srv.debugLog("jawsauth: scheduled auth timer",
                "session_prefix", debugSessionPrefix(sess),
                "mode", mode,
                "expiry", expiry,
                "delay", delay,
                "replaced_existing", replaced,
        )
}

func (srv *Server) emitLoginEvent(sess *jaws.Session, hr *http.Request) {
        srv.mu.Lock()
        pending := srv.loginEvents[sess]
        if pending == nil {
                srv.mu.Unlock()
                return
        }
        if pending.cleared {
                delete(srv.loginEvents, sess)
                srv.mu.Unlock()
                return
        }
        srv.mu.Unlock()
        defer func() {
                srv.mu.Lock()
                delete(srv.loginEvents, sess)
                cleared := pending.cleared
                srv.mu.Unlock()
                if cleared && srv.LoginEvent != nil && srv.LogoutEvent != nil {
                        srv.LogoutEvent(sess, nil)
                }
        }()
        if srv.LoginEvent != nil {
                srv.LoginEvent(sess, hr)
        }
}

func (srv *Server) setSessionAuthFromToken(ctx context.Context, sess *jaws.Session, tokenSource oauth2.TokenSource, token *oauth2.Token, minExpiry time.Time, entry *authTimerState) error {
        if srv == nil || srv.idTokenVerifier == nil {
                return ErrOAuth2NotConfigured
        }
        var raw string
        if token != nil {
                raw, _ = token.Extra("id_token").(string)
        }
        idToken, claims, err := srv.verifyIDToken(ctx, raw)
        if err != nil {
                return err
        }
        if !minExpiry.IsZero() && !idToken.Expiry.After(minExpiry) {
                return errOIDC{kind: ErrOIDCInvalidIDToken, cause: errOIDCStaleIDToken}
        }
        srv.addUserInfoClaims(ctx, claims, tokenSource)
        return srv.storeSessionAuthClaims(sess, claims, tokenSource, idToken.Expiry, entry)
}

func (srv *Server) refreshSessionAuth(ctx context.Context, sess *jaws.Session, minExpiry time.Time, entry *authTimerState) (err error) {
        sessionPrefix := debugSessionPrefix(sess)
        srv.debugLog("jawsauth: refresh session auth started",
                "session_prefix", sessionPrefix,
                "min_expiry", minExpiry,
                "timer_entry", entry != nil,
                "entry_expiry", authTimerEntryExpiry(entry),
        )
        err = ErrOAuth2NotConfigured
        if srv != nil && sess != nil && srv.oauth2cfg != nil && srv.idTokenVerifier != nil {
                tokenSource, _ := sess.Get(srv.SessionTokenKey).(oauth2.TokenSource)
                err = ErrOIDCMissingIDToken
                if tokenSource != nil {
                        authctx := srv.oauth2Context(ctx)
                        var token *oauth2.Token
                        srv.debugLog("jawsauth: requesting token from stored token source", "session_prefix", sessionPrefix)
                        if token, err = tokenSource.Token(); err == nil {
                                srv.debugLog("jawsauth: stored token source returned token", append([]any{"session_prefix", sessionPrefix}, tokenDebugAttrs(token)...)...)
                                err = srv.setSessionAuthFromToken(authctx, sess, tokenSource, token, minExpiry, entry)
                                if err == nil {
                                        srv.debugLog("jawsauth: stored token refreshed session auth", "session_prefix", sessionPrefix)
                                } else {
                                        srv.debugErrorLog("jawsauth: stored token did not refresh session auth", err, "session_prefix", sessionPrefix)
                                }
                                if err != nil && token != nil && token.RefreshToken != "" && !errors.Is(err, errAuthTimerStale) {
                                        srv.debugErrorLog("jawsauth: forcing refresh with refresh token", err, "session_prefix", sessionPrefix)
                                        tokenSource = srv.oauth2cfg.TokenSource(authctx, &oauth2.Token{
                                                RefreshToken: token.RefreshToken,
                                        })
                                        if token, err = tokenSource.Token(); err == nil {
                                                srv.debugLog("jawsauth: forced refresh returned token", append([]any{"session_prefix", sessionPrefix}, tokenDebugAttrs(token)...)...)
                                                err = srv.setSessionAuthFromToken(authctx, sess, tokenSource, token, minExpiry, entry)
                                                if err == nil {
                                                        srv.debugLog("jawsauth: forced refresh updated session auth", "session_prefix", sessionPrefix)
                                                } else {
                                                        srv.debugErrorLog("jawsauth: forced refresh did not update session auth", err, "session_prefix", sessionPrefix)
                                                }
                                        } else {
                                                srv.debugErrorLog("jawsauth: forced refresh token source failed", err, "session_prefix", sessionPrefix)
                                        }
                                }
                        } else {
                                srv.debugErrorLog("jawsauth: stored token source failed", err, "session_prefix", sessionPrefix)
                        }
                } else {
                        srv.debugLog("jawsauth: refresh session auth missing token source", "session_prefix", sessionPrefix)
                }
        } else {
                srv.debugLog("jawsauth: refresh session auth not configured",
                        "session_prefix", sessionPrefix,
                        "server_nil", srv == nil,
                        "session_nil", sess == nil,
                        "oauth2_configured", srv != nil && srv.oauth2cfg != nil,
                        "id_token_verifier_configured", srv != nil && srv.idTokenVerifier != nil,
                )
        }
        return
}

// Caller holds srv.mu across the auth writes and timer replacement.
func (srv *Server) scheduleSessionAuthTimerLocked(sess *jaws.Session, expiry time.Time) (delay time.Duration, replaced bool) {
        remaining := time.Until(expiry)
        delay = remaining
        if srv.profileSubject == nil {
                // Short-lived tokens wait half their remaining lifetime to avoid refresh loops.
                delay = max(remaining-authRefreshSkew, remaining/2)
        }
        delay = min(max(delay, 0), authSessionCheckInterval)
        entry := &authTimerState{expiry: expiry}
        if srv.authTimers == nil {
                srv.authTimers = make(map[uint64]*authTimerState)
        }
        if srv.authTimerAfterFunc == nil {
                srv.authTimerAfterFunc = realAuthTimerAfterFunc
        }
        if old := srv.authTimers[sess.ID()]; old != nil && old.timer != nil {
                replaced = true
                old.timer.Stop()
        }
        srv.authTimers[sess.ID()] = entry
        entry.timer = srv.authTimerAfterFunc(delay, func() {
                srv.handleSessionAuthTimer(sess, entry)
        })
        return
}

func (srv *Server) sessionAuthTimerCurrent(sess *jaws.Session, entry *authTimerState) (current bool) {
        if srv != nil && sess != nil && entry != nil {
                srv.mu.Lock()
                current = srv.authTimers[sess.ID()] == entry
                srv.mu.Unlock()
        }
        return
}

func (srv *Server) stopSessionAuthTimerLocked(sess *jaws.Session, entry *authTimerState) bool {
        old := srv.authTimers[sess.ID()]
        if entry != nil && old != entry {
                return false
        }
        delete(srv.authTimers, sess.ID())
        if old != nil && old.timer != nil {
                old.timer.Stop()
        }
        return true
}

func (srv *Server) rearmSessionAuthTimer(sess *jaws.Session, entry *authTimerState, delay time.Duration) bool {
        srv.mu.Lock()
        defer srv.mu.Unlock()
        if srv.authTimers[sess.ID()] != entry {
                return false
        }
        entry.timer = srv.authTimerAfterFunc(delay, func() {
                srv.handleSessionAuthTimer(sess, entry)
        })
        return true
}

func (srv *Server) handleSessionAuthTimer(sess *jaws.Session, entry *authTimerState) {
        if srv.sessionAuthTimerCurrent(sess, entry) {
                remaining := time.Until(srv.authTimerDue(entry.expiry))
                if !sessionAlive(sess) || (srv.profileSubject != nil && remaining <= 0) {
                        srv.clearSessionAuth(sess, nil, entry, true)
                        return
                }
                if remaining > 0 {
                        srv.rearmSessionAuthTimer(sess, entry, min(remaining, authSessionCheckInterval))
                        return
                }
                current, present := srv.sessionAuthStatus(sess, time.Now)
                srv.debugLog("jawsauth: auth refresh timer fired",
                        "session_prefix", debugSessionPrefix(sess),
                        "entry_expiry", authTimerEntryExpiry(entry),
                        "session_current", current,
                        "session_present", present,
                )
                err := srv.refreshSessionAuth(context.Background(), sess, entry.expiry, entry)
                if err != nil {
                        if errors.Is(err, errAuthTimerStale) {
                                srv.debugErrorLog("jawsauth: auth refresh timer became stale", err, "session_prefix", debugSessionPrefix(sess))
                                return
                        }
                        current, present = srv.sessionAuthStatus(sess, time.Now)
                        if current && present {
                                retryDelay := max(time.Until(entry.expiry), 0)
                                if srv.rearmSessionAuthTimer(sess, entry, retryDelay) {
                                        srv.debugErrorLog("jawsauth: auth refresh timer failed; keeping current auth", err,
                                                "session_prefix", debugSessionPrefix(sess),
                                                "entry_expiry", authTimerEntryExpiry(entry),
                                                "session_current", current,
                                                "session_present", present,
                                                "retry_delay", retryDelay,
                                        )
                                        return
                                }
                        }
                        srv.debugErrorLog("jawsauth: auth refresh timer failed; clearing auth", err,
                                "session_prefix", debugSessionPrefix(sess),
                                "entry_expiry", authTimerEntryExpiry(entry),
                                "session_current", current,
                                "session_present", present,
                        )
                        _ = srv.Jaws.Log(err)
                        srv.clearSessionAuth(sess, nil, entry, true)
                } else {
                        srv.debugLog("jawsauth: auth refresh timer completed", "session_prefix", debugSessionPrefix(sess))
                }
        } else if sess != nil {
                srv.debugLog("jawsauth: stale auth refresh timer ignored",
                        "session_prefix", debugSessionPrefix(sess),
                        "entry_expiry", authTimerEntryExpiry(entry),
                )
        }
}

func (srv *Server) authTimerDue(expiry time.Time) time.Time {
        if srv.profileSubject != nil {
                return expiry
        }
        return expiry.Add(-authRefreshSkew)
}

// pendingOrdinaryLoginLocked requires srv.mu.
func (srv *Server) pendingOrdinaryLoginLocked(sess *jaws.Session, minimumRemaining time.Duration) bool {
        if flow := srv.oauthCallbacks[sess]; flow.canceled != nil {
                return !flow.link
        }
        flow, _ := sess.Get(srv.sessionKey(oauth2FlowKey)).(pendingOAuthFlow)
        return flow.state != "" && !flow.link && time.Until(flow.expiry) > minimumRemaining
}

func (srv *Server) clearSessionOAuthFlowLocked(sess *jaws.Session) {
        if flow := srv.oauthCallbacks[sess]; flow.canceled != nil {
                close(flow.canceled)
                delete(srv.oauthCallbacks, sess)
        }
        sess.Set(srv.sessionKey(oauth2FlowKey), nil)
}

// Logout clears this server's authentication state for sess.
//
// It stops the auth timer and clears claims, token source, email and its
// verification flag, expiry, and in-flight OAuth flow keys, then marks the
// session dirty. If auth was present, it reloads its pages and may call
// [Server.LogoutEvent] under [EventFunc]'s ordering rules. It returns false for
// a nil receiver or session, and true otherwise. The hr argument may be nil.
//
// It performs no HTTP redirect. An HTTP handler can build its own post-logout
// response. A JaWS event handler cannot use [jaws.Request.Redirect] after Logout
// queues Reload, which closes the connection; use an HTTP logout endpoint instead.
// It takes SessionRotationLock, if set, and releases it before LogoutEvent runs.
func (srv *Server) Logout(sess *jaws.Session, hr *http.Request) (cleared bool) {
        return srv.clearSessionAuth(sess, hr, nil, false)
}

func (srv *Server) clearSessionAuth(sess *jaws.Session, hr *http.Request, entry *authTimerState, preservePendingLogin bool) (cleared bool) {
        if srv == nil || sess == nil {
                return
        }
        rotationLock := srv.SessionRotationLock
        if rotationLock != nil {
                rotationLock.Lock()
        }
        srv.mu.Lock()
        pending := srv.loginEvents[sess]
        hadAuth := srv.authTimers[sess.ID()] != nil
        if !hadAuth {
                _, hadAuth = srv.sessionAuthStatus(sess, nil)
        }
        cleared = srv.stopSessionAuthTimerLocked(sess, entry)
        var requests []*jaws.Request
        if cleared {
                // Auth expiry must not cancel an ordinary login already in progress.
                // Explicit logout and link flows still revoke their pending callbacks.
                if !preservePendingLogin || !srv.pendingOrdinaryLoginLocked(sess, 0) {
                        srv.clearSessionOAuthFlowLocked(sess)
                }
                sess.Set(srv.SessionKey, nil)
                sess.Set(srv.SessionTokenKey, nil)
                sess.Set(srv.sessionKey(oauth2AuthExpiryKey), nil)
                sess.Set(srv.SessionEmailKey, nil)
                sess.Set(srv.SessionEmailVerifiedKey, nil)
                if hadAuth {
                        requests = sess.Requests()
                        if pending != nil {
                                pending.cleared = true
                        }
                }
        }
        srv.mu.Unlock()
        if rotationLock != nil {
                rotationLock.Unlock()
        }
        if cleared {
                if hadAuth {
                        reloadAuthRequests(requests)
                        if (pending == nil || srv.LoginEvent == nil) && srv.LogoutEvent != nil {
                                srv.LogoutEvent(sess, hr)
                        }
                }
                srv.Jaws.Dirty(sess)
                srv.debugLog("jawsauth: cleared session auth",
                        "session_prefix", debugSessionPrefix(sess),
                        "request_present", hr != nil,
                        "timer_entry", entry != nil,
                        "entry_expiry", authTimerEntryExpiry(entry),
                )
        }
        return
}

func reloadAuthRequests(requests []*jaws.Request) {
        for _, rq := range requests {
                rq.Reload()
        }
}

// Config holds the OIDC/OAuth2 settings used by New and NewDebug to construct a Server.
//
// RedirectURL, Issuer and ClientID are required. Setting ProfileSubject selects
// profile mode and requires AuthURL, TokenURL and UserInfoURL. Otherwise OIDC
// mode discovers endpoints; those three URLs may override them.
type Config struct {
        RedirectURL string // required. e.g. "https://application.example.com/oauth2/callback"
        Issuer      string // required. e.g. "https://login.microsoftonline.com/00000000-0000-0000-0000-000000000000/v2.0"
        // Namespace isolates providers sharing one JaWS instance. New appends it to
        // the four exported Server session-key fields; use a distinct value per provider.
        Namespace   string
        AuthURL     string
        TokenURL    string
        UserInfoURL string
        // AllowInsecureIssuer permits HTTP issuer and provider endpoints and is only for tests/dev.
        AllowInsecureIssuer bool
        // HTTPClient handles discovery, JWKS, token exchange/refresh and UserInfo.
        // Requests cap its timeout at 15 seconds. Discovery/JWKS follow only HTTPS
        // redirects unless AllowInsecureIssuer is set; token and UserInfo follow none.
        // A request's oauth2.HTTPClient can override code exchange and UserInfo;
        // background refresh uses HTTPClient.
        HTTPClient   *http.Client
        Scopes       []string // optional; OIDC adds "openid" and "email"; profile mode adds none
        ClientID     string
        ClientSecret string
        // ProfileSubject enables OAuth2 profile mode and returns a stable, non-empty
        // subject from the authenticated profile. It may map provider verification
        // data to claims["email_verified"] for admin checks. Profile JSON numbers
        // arrive as json.Number.
        ProfileSubject func(map[string]any) (string, error)
}

func requireLen(k string, n int) (err error) {
        if n < 1 {
                err = errConfig{field: k, cause: ErrConfigMissingValue}
        }
        return
}

func requireStr(k, u string) error {
        return requireLen(k, len(strings.TrimSpace(u)))
}

func validateUrl(k, value string, optional bool) (*url.URL, error) {
        if optional && value == "" {
                return nil, nil
        }
        if err := requireStr(k, value); err != nil {
                return nil, err
        }
        parsed, err := url.Parse(value)
        if err != nil {
                return nil, errConfig{field: k, cause: err}
        }
        if !parsed.IsAbs() {
                return nil, errConfig{field: k, cause: ErrConfigURLNotAbsolute}
        }
        if parsed.Hostname() == "" {
                return nil, errConfig{field: k, cause: ErrConfigURLMissingHost}
        }
        return parsed, nil
}

func validateEndpoint(field, value string, optional, allowInsecure bool) error {
        parsed, err := validateUrl(field, value, optional)
        if err != nil || parsed == nil {
                return err
        }
        if parsed.Scheme != "https" && (!allowInsecure || parsed.Scheme != "http") {
                return errConfig{field: field, cause: ErrConfigEndpointMustBeHTTPS}
        }
        return nil
}

// Validate checks whether cfg contains usable OIDC/OAuth2 settings.
//
// It checks required fields and absolute URLs with hosts. Configured issuer and
// provider endpoints require HTTPS unless AllowInsecureIssuer permits HTTP.
// Returned failures match [ErrConfig].
func (cfg *Config) Validate() error {
        if _, err := validateUrl("RedirectURL", cfg.RedirectURL, false); err != nil {
                return err
        }
        issuer, err := validateUrl("Issuer", cfg.Issuer, false)
        if err != nil {
                return err
        }
        if issuer.Scheme != "https" && (!cfg.AllowInsecureIssuer || issuer.Scheme != "http") {
                return errConfig{field: "Issuer", cause: ErrConfigIssuerMustBeHTTPS}
        }
        for _, field := range []struct{ name, value string }{
                {"AuthURL", cfg.AuthURL}, {"TokenURL", cfg.TokenURL}, {"UserInfoURL", cfg.UserInfoURL},
        } {
                if err := validateEndpoint(field.name, field.value, cfg.ProfileSubject == nil, cfg.AllowInsecureIssuer); err != nil {
                        return err
                }
        }
        return requireStr("ClientID", cfg.ClientID)
}

func overrideStr(a *string, b string) {
        if b != "" {
                *a = b
        }
}

func ensureScopes(rawScopes []string, required ...string) (scopes []string) {
        scopes = append(scopes, required...)
        for _, set := range rawScopes {
                for scope := range strings.FieldsSeq(set) {
                        scopes = append(scopes, scope)
                }
        }
        slices.Sort(scopes)
        scopes = slices.Compact(scopes)
        return
}

func boundedProviderClient(client *http.Client) (bounded http.Client) {
        if client == nil {
                client = http.DefaultClient
        }
        bounded = *client
        if bounded.Timeout <= 0 || bounded.Timeout > oauth2ProviderTimeout {
                bounded.Timeout = oauth2ProviderTimeout
        }
        return
}

func (cfg *Config) buildContext(ctx context.Context, overrideUrl string) (oauth2cfg *oauth2.Config, userInfoURL string, verifier *oidc.IDTokenVerifier, err error) {
        if err = cfg.Validate(); err == nil {
                redir, _ := url.Parse(cfg.RedirectURL)
                if u, parseErr := url.Parse(overrideUrl); parseErr == nil {
                        overrideStr(&redir.Scheme, u.Scheme)
                        overrideStr(&redir.Host, u.Host)
                }
                if cfg.ProfileSubject != nil {
                        scopes := ensureScopes(cfg.Scopes)
                        oauth2cfg = &oauth2.Config{
                                ClientID: cfg.ClientID, ClientSecret: cfg.ClientSecret,
                                Endpoint:    oauth2.Endpoint{AuthURL: cfg.AuthURL, TokenURL: cfg.TokenURL},
                                RedirectURL: redir.String(), Scopes: scopes,
                        }
                        return oauth2cfg, cfg.UserInfoURL, nil, nil
                }
                clientCopy := boundedProviderClient(cfg.HTTPClient)
                if !cfg.AllowInsecureIssuer {
                        checkRedirect := clientCopy.CheckRedirect
                        clientCopy.CheckRedirect = func(req *http.Request, via []*http.Request) error {
                                if req.URL.Scheme != "https" {
                                        return errors.New("oidc redirect must use https")
                                }
                                if checkRedirect != nil {
                                        return checkRedirect(req, via)
                                }
                                if len(via) >= 10 {
                                        return errors.New("stopped after 10 redirects")
                                }
                                return nil
                        }
                }
                ctx = context.WithValue(ctx, oauth2.HTTPClient, &clientCopy)

                var provider *oidc.Provider
                if provider, err = oidc.NewProvider(ctx, cfg.Issuer); wrapOIDC(ErrOIDCDiscovery, &err) == nil {
                        var metadata struct {
                                AuthorizationEndpoint string `json:"authorization_endpoint"`
                                TokenEndpoint         string `json:"token_endpoint"`
                                UserinfoEndpoint      string `json:"userinfo_endpoint"`
                                JWKSURI               string `json:"jwks_uri"`
                        }
                        if err = provider.Claims(&metadata); wrapOIDC(ErrOIDCProviderMetadata, &err) == nil {
                                authURL := cmp.Or(cfg.AuthURL, metadata.AuthorizationEndpoint)
                                tokenURL := cmp.Or(cfg.TokenURL, metadata.TokenEndpoint)
                                userInfoURL = cmp.Or(cfg.UserInfoURL, metadata.UserinfoEndpoint)
                                for _, field := range []struct {
                                        name, value string
                                        optional    bool
                                }{
                                        {"authorization_endpoint", authURL, false},
                                        {"token_endpoint", tokenURL, false},
                                        {"userinfo_endpoint", userInfoURL, true},
                                        {"jwks_uri", metadata.JWKSURI, false},
                                } {
                                        if err = validateEndpoint(field.name, field.value, field.optional, cfg.AllowInsecureIssuer); wrapOIDC(ErrOIDCProviderMetadata, &err) != nil {
                                                return
                                        }
                                }
                                oauth2cfg = &oauth2.Config{
                                        ClientID:     cfg.ClientID,
                                        ClientSecret: cfg.ClientSecret,
                                        Endpoint: oauth2.Endpoint{
                                                AuthURL:  authURL,
                                                TokenURL: tokenURL,
                                        },
                                        RedirectURL: redir.String(),
                                        Scopes:      ensureScopes(cfg.Scopes, "openid", "email"),
                                }
                                verifier = provider.Verifier(&oidc.Config{ClientID: cfg.ClientID})
                        }
                }
        }
        return
}

func oauth2RequestDebugAttrs(req *http.Request) (attrs []any) {
        attrs = []any{"request_nil", req == nil}

func requestBodyDebugAttrs(contentType string, body []byte) (attrs []any) {
        attrs = []any{"request_body_len", len(body)}
        mediaType := strings.ToLower(strings.TrimSpace(strings.Split(contentType, ";")[0]))
        if len(body) != 0 {
                switch {
                case mediaType == "application/x-www-form-urlencoded":
                        values, err := url.ParseQuery(string(body))
                        if err == nil {
                                attrs = append(attrs, "request_form_data", redactDebugQuery(values))
                        } else {
                                attrs = append(attrs, "request_body_parse_error", err.Error())
                                attrs = append(attrs, redactDebugBodyText(body)...)
                        }
                case mediaType == "application/json" || strings.HasSuffix(mediaType, "+json"):
                        var value any
                        err := json.Unmarshal(body, &value)
                        if err == nil {
                                attrs = append(attrs, "request_json_data", redactDebugJSONValue("", value))
                        } else {
                                attrs = append(attrs, "request_body_parse_error", err.Error())
                                attrs = append(attrs, redactDebugBodyText(body)...)
                        }
                default:
                        attrs = append(attrs, redactDebugBodyText(body)...)
                }
        }
        return
}

func redactDebugBodyText(body []byte) (attrs []any) {
        text := string(body)
        if len(text) > debugRequestBodyLimit {
                text = text[:debugRequestBodyLimit]
                attrs = append(attrs, "request_body_truncated", true)
        }
        attrs = append(attrs, "request_body", text)
        return
}

func redactDebugJSONValue(key string, value any) (redacted any) {
        redacted = value
        if isSensitiveDebugKey(key) {
                redacted = debugRedactedValue
        } else {
                switch v := value.(type) {
                case map[string]any:
                        m := make(map[string]any, len(v))
                        for k, item := range v {
                                m[k] = redactDebugJSONValue(k, item)
                        }
                        redacted = m
                case []any:
                        sl := make([]any, len(v))
                        for i, item := range v {
                                sl[i] = redactDebugJSONValue(key, item)
                        }
                        redacted = sl
                }
        }
        return
}

func isSensitiveDebugKey(key string) (sensitive bool) {
        key = strings.ToLower(key)
        for _, marker := range []string{
                "authorization",
                "assertion",
                "code",
                "cookie",
                "password",
                "secret",
                "token",
                "verifier",
        } {
                if strings.Contains(key, marker) {
                        sensitive = true
                        return
                }
        }
        return
}

func errorDebugAttrs(err error) (attrs []any) {
        attrs = []any{"err", err}
        if err != nil {
                var errChain []string
                var errChainTypes []string
                appendErrorChainAttrs(&errChain, &errChainTypes, err)
                attrs = append(attrs,
                        "err_type", fmt.Sprintf("%T", err),
                        "err_chain", errChain,
                        "err_chain_types", errChainTypes,
                )
                if classes := errorDebugClasses(err); len(classes) != 0 {
                        attrs = append(attrs, "err_classes", classes)
                }
                var callbackErr *OAuth2CallbackError
                if errors.As(err, &callbackErr) && callbackErr != nil {
                        attrs = append(attrs,
                                "oauth2_callback_error", true,
                                "oauth2_callback_error_code", callbackErr.Code,
                                "oauth2_callback_error_description", callbackErr.Description,
                                "oauth2_callback_error_uri", callbackErr.URI,
                        )
                }
                var retrieveErr *oauth2.RetrieveError
                if errors.As(err, &retrieveErr) && retrieveErr != nil {
                        attrs = append(attrs,
                                "oauth2_retrieve_error", true,
                                "oauth2_retrieve_error_code", retrieveErr.ErrorCode,
                                "oauth2_retrieve_error_description", retrieveErr.ErrorDescription,
                                "oauth2_retrieve_error_uri", retrieveErr.ErrorURI,
                        )
                        if retrieveErr.Response != nil {
                                attrs = append(attrs,
                                        "oauth2_response_status", retrieveErr.Response.Status,
                                        "oauth2_response_status_code", retrieveErr.Response.StatusCode,
                                )
                        }
                        if len(retrieveErr.Body) != 0 {
                                body := string(retrieveErr.Body)
                                if len(body) > debugErrorBodyLimit {
                                        body = body[:debugErrorBodyLimit]
                                        attrs = append(attrs, "oauth2_response_body_truncated", true)
                                }
                                attrs = append(attrs, "oauth2_response_body", body)
                        }
                }
        }
        return
}

                appendErrorChainAttrs(errChain, errChainTypes, errors.Unwrap(err))
        }
}

        Code        string // OAuth2 error code from the callback.
        Description string // Optional error description from the callback.
        URI         string // Optional URI with details about the callback error.
}

func (err *OAuth2CallbackError) Error() string {
        if err == nil {
                return ErrOAuth2Callback.Error()
        }
        var sb strings.Builder
        sb.WriteString(ErrOAuth2Callback.Error())
        if s := strings.TrimSpace(err.Code); s != "" {
                sb.WriteString(": ")
                sb.WriteString(s)
        }
        if s := strings.TrimSpace(err.Description); s != "" {
                sb.WriteString(": ")
                sb.WriteString(s)
        }
        if s := strings.TrimSpace(err.URI); s != "" {
                sb.WriteString(" (")
                sb.WriteString(s)
                sb.WriteString(")")
        }
        return sb.String()
}

func (err *OAuth2CallbackError) Is(target error) bool {
        return target == ErrOAuth2Callback
}

func oauth2CallbackError(statusCode int, hr *http.Request) (nextStatusCode int, err error) {
        nextStatusCode = statusCode
        if s := strings.TrimSpace(hr.FormValue("error")); s != "" {
                callbackErr := &OAuth2CallbackError{
                        Code:        s,
                        Description: strings.TrimSpace(hr.FormValue("error_description")),
                        URI:         strings.TrimSpace(hr.FormValue("error_uri")),
                }
                nextStatusCode = http.StatusBadRequest
                switch callbackErr.Code {
                case "access_denied":
                        nextStatusCode = http.StatusForbidden
                }
                err = callbackErr
        }
        return
}

// CurrentClaims returns this provider's live authentication claims.
//
// While an in-place login notification is pending, it returns nil.
// It is useful when several providers share JaWS.
// Treat the returned map as read-only. The session may close, and an in-place
// login may change its identity or permissions after this call. Recheck the
// session and claims before protected writes.
func (srv *Server) CurrentClaims(sess *jaws.Session) map[string]any {
        if srv == nil || srv.Jaws == nil || srv.SessionKey == "" || sess == nil || sess.Jaws() != srv.Jaws {
                return nil
        }
        srv.mu.Lock()
        defer srv.mu.Unlock()
        if !srv.RotateSessionOnLogin && srv.loginEvents[sess] != nil {
                return nil
        }
        if current, _ := srv.sessionAuthStatus(sess, time.Now); !current {
                return nil
        }
        claims, _ := sess.Get(srv.SessionKey).(map[string]any)
        if !sessionAlive(sess) {
                return nil
        }
        return claims
}

const oauth2FlowKey = "oauth2flow"
const oauth2AuthExpiryKey = "oauth2authexpiry"

const oauth2FlowLifetime = 10 * time.Minute
const oauth2ProviderTimeout = 15 * time.Second
const profileAuthLifetime = 24 * time.Hour

// Allow provider I/O after the flow lifetime, with one second of extra grace.
const oauth2FlowGrace = oauth2FlowLifetime + oauth2ProviderTimeout + time.Second

type pendingOAuthFlow struct {
        owner    *Server
        state    string
        expiry   time.Time
        verifier string
        nonce    string
        referrer string
        link     bool
        canceled chan struct{}
}

func (flow pendingOAuthFlow) revoked() bool {
        select {
        case <-flow.canceled:
                return true
        default:
                return false
        }
}

func (srv *Server) takeOAuthFlow(sess *jaws.Session, gotState string) (flow pendingOAuthFlow, err error) {
        srv.mu.Lock()
        defer srv.mu.Unlock()
        flow, _ = sess.Get(srv.sessionKey(oauth2FlowKey)).(pendingOAuthFlow)
        if flow.state == "" {
                return flow, ErrOAuth2MissingState
        }
        if gotState != flow.state || flow.owner != srv {
                return flow, ErrOAuth2WrongState
        }
        srv.clearSessionOAuthFlowLocked(sess)
        if !time.Now().Before(flow.expiry) {
                return flow, ErrOAuth2ExpiredState
        }
        flow.canceled = make(chan struct{})
        if srv.oauthCallbacks == nil {
                srv.oauthCallbacks = make(map[*jaws.Session]pendingOAuthFlow)
        }
        srv.oauthCallbacks[sess] = flow
        return flow, nil
}

func (srv *Server) finishOAuthFlow(sess *jaws.Session, flow pendingOAuthFlow) {
        srv.mu.Lock()
        if srv.oauthCallbacks[sess].canceled == flow.canceled {
                delete(srv.oauthCallbacks, sess)
        }
        srv.mu.Unlock()
}

// maxRedirectTargetLen bounds return targets retained in pre-auth sessions.
const maxRedirectTargetLen = 8192

func normalizeHost(hostport string) (normalized string) {
        normalized = strings.TrimSpace(hostport)
        if normalized != "" {
                if strings.HasPrefix(normalized, "[") && strings.HasSuffix(normalized, "]") {
                        normalized += ":0"
                }
                if strings.Contains(normalized, ":") {
                        h, _, err := net.SplitHostPort(normalized)
                        if err != nil {
                                return ""
                        }
                        normalized = h
                }
                normalized = strings.ToLower(strings.TrimSuffix(normalized, "."))
        }
        return
}

func sanitizeRedirectTarget(requestHost, location string) (sanitized string) {
        if trimmed := strings.TrimSpace(location); trimmed != "" {
                if u, err := url.Parse(trimmed); err == nil {
                        if u.IsAbs() {
                                if host := normalizeHost(requestHost); host != "" {
                                        if normalizeHost(u.Host) == host {
                                                sanitized = u.RequestURI()
                                        }
                                }
                        } else {
                                sanitized = trimmed
                        }
                }
        }
        sanitized = strings.TrimSpace(sanitized)
        sanitized = "/" + strings.TrimLeft(sanitized, "/\\")
        if len(sanitized) > maxRedirectTargetLen {
                sanitized = "/"
        }
        return
}

func (srv *Server) begin(hr *http.Request) (location string) {
        if location = strings.TrimSpace(hr.Referer()); location == "" {
                location = hr.RequestURI
        }
        location = srv.returnTarget(hr, location)
        return
}

func (srv *Server) returnTarget(hr *http.Request, location string) string {
        location = sanitizeRedirectTarget(hr.Host, location)
        if target, err := url.Parse(location); err == nil {
                if _, handled := srv.HandledPaths[callbackPathFromURL(target)]; handled {
                        location = "/"
                }
        }
        return location
}

// HandleLogin begins the configured OIDC or OAuth2 profile login flow.
//
// For GET requests it stores the state, PKCE verifier and, for OIDC, nonce in the
// session, then responds with a 302 redirect to the provider's authorization URL.
// It replaces any pending flow for this server and session, canceling a running
// callback or link, unless an in-place login notification is pending; then the
// default response is 409.
// If no usable session is available, it returns 503 with Retry-After instead.
// Non-GET requests receive 405 with Allow: GET.
func (srv *Server) HandleLogin(hw http.ResponseWriter, hr *http.Request) {
        if hr.Method != http.MethodGet {
                hw.Header().Set("Allow", http.MethodGet)
                srv.writeResult(hw, http.StatusMethodNotAllowed, nil)
                return
        }
        srv.startOAuth(hw, hr, false, false)
}

// HandleLink begins a provider flow for linking to the caller's existing
// identity. The application must authorize this POST before calling it and
// recheck that identity in LinkEvent when the provider returns. On success it
// responds with a 303 redirect to the provider. It returns 405 with Allow: POST
// for other methods, 403 for a cross-origin or missing session, and 500 when
// linking is not configured. The default response is 409 while an in-place login
// notification is pending. Otherwise, starting a link replaces any pending
// flow for this server and session, including a running login callback.
func (srv *Server) HandleLink(hw http.ResponseWriter, hr *http.Request) {
        switch {
        case hr.Method != http.MethodPost:
                hw.Header().Set("Allow", http.MethodPost)
                srv.writeResult(hw, http.StatusMethodNotAllowed, nil)
        case linkOriginCheck.Check(hr) != nil:
                srv.writeResult(hw, http.StatusForbidden, ErrOAuth2InvalidOrigin)
        case srv.oauth2cfg == nil || srv.LinkEvent == nil:
                srv.writeResult(hw, http.StatusInternalServerError, ErrOAuth2NotConfigured)
        default:
                srv.startOAuth(hw, hr, true, false)
        }
}

var linkOriginCheck = http.NewCrossOriginProtection()

func (srv *Server) startOAuth(hw http.ResponseWriter, hr *http.Request, link, fromWrapper bool) {
        oauth2cfg := srv.oauth2cfg
        var location string
        switch {
        case fromWrapper:
                location = srv.returnTarget(hr, hr.RequestURI)
        case link && strings.TrimSpace(hr.Referer()) == "":
                location = "/"
        default:
                location = srv.begin(hr)
        }
        if oauth2cfg != nil {
                flow, sess, status := srv.prepareOAuthFlow(hw, hr, link, fromWrapper, location)
                if status != 0 {
                        failure := ErrOAuth2MissingSession
                        switch status {
                        case http.StatusConflict:
                                failure = ErrOAuth2LoginInProgress
                                if srv.LoginFailed != nil {
                                        email, _ := sess.Get(srv.SessionEmailKey).(string)
                                        if srv.LoginFailed(hw, hr, status, failure, email) {
                                                return
                                        }
                                }
                        case http.StatusServiceUnavailable:
                                hw.Header().Set("Retry-After", "60")
                        }
                        srv.writeResult(hw, status, failure)
                        return
                }
                authOptions := append([]oauth2.AuthCodeOption{}, srv.Options...)
                if flow.nonce != "" {
                        authOptions = append(authOptions, oidc.Nonce(flow.nonce))
                }
                authOptions = append(authOptions, oauth2.S256ChallengeOption(flow.verifier))
                location = oauth2cfg.AuthCodeURL(flow.state, authOptions...)
        }
        hw.Header().Set("Location", location)
        statusCode := http.StatusFound
        if link {
                statusCode = http.StatusSeeOther
        }
        SetHeaders(hw, srv.ishttps)
        hw.WriteHeader(statusCode)
}

func (srv *Server) prepareOAuthFlow(hw http.ResponseWriter, hr *http.Request, link, fromWrapper bool, location string) (flow pendingOAuthFlow, sess *jaws.Session, status int) {
        failStatus := http.StatusServiceUnavailable
        if link {
                failStatus = http.StatusForbidden
        }
        rotationLock := srv.SessionRotationLock
        if rotationLock != nil {
                rotationLock.Lock()
                defer rotationLock.Unlock()
        }
        sess = srv.Jaws.GetSession(hr)
        if sess == nil && !link {
                sess = srv.Jaws.NewSession(hw, hr)
        }
        if sess == nil {
                return flow, nil, failStatus
        }
        srv.mu.Lock()
        defer srv.mu.Unlock()
        if !srv.RotateSessionOnLogin && srv.loginEvents[sess] != nil {
                return flow, sess, http.StatusConflict
        }
        if fromWrapper && srv.pendingOrdinaryLoginLocked(sess, 2*time.Minute) {
                // Reissue one pending transaction without replacing the first tab's state.
                flow, _ = sess.Get(srv.sessionKey(oauth2FlowKey)).(pendingOAuthFlow)
                if flow.state == "" || flow.owner != srv {
                        status = http.StatusConflict
                }
                return
        }
        flow = pendingOAuthFlow{
                owner: srv,
                state: rand.Text(), expiry: time.Now().Add(oauth2FlowLifetime),
                link: link, verifier: oauth2.GenerateVerifier(), referrer: location,
        }
        if srv.profileSubject == nil {
                flow.nonce = rand.Text()
        }
        if !sess.ExtendGrace(oauth2FlowGrace) {
                status = failStatus
                return
        }
        srv.clearSessionOAuthFlowLocked(sess)
        sess.Set(srv.sessionKey(oauth2FlowKey), flow)
        return
}

// HandleLogout clears this server's authentication in the session and redirects.
//
// For GET requests it clears auth and responds with a 302 redirect to the
// sanitized referrer (or "/"). LogoutEvent follows [EventFunc]'s ordering rules.
// Non-GET requests receive 405 with Allow: GET.
func (srv *Server) HandleLogout(hw http.ResponseWriter, hr *http.Request) {
        statusCode := http.StatusMethodNotAllowed
        if hr.Method == http.MethodGet {
                location := srv.begin(hr)
                if sess := srv.Jaws.GetSession(hr); sess != nil {
                        srv.clearSessionAuth(sess, hr, nil, false)
                }
                hw.Header().Set("Location", location)
                statusCode = http.StatusFound
        } else {
                hw.Header().Set("Allow", http.MethodGet)
        }
        SetHeaders(hw, srv.ishttps)
        hw.WriteHeader(statusCode)
}

// SetHeaders writes the HTTP response headers for all OAuth endpoint responses.

// It defaults to DefaultSetHeaders and may be reassigned at startup to customize
// headers; it must not be reassigned concurrently with serving requests.
var SetHeaders = DefaultSetHeaders

func DefaultSetHeaders(hw http.ResponseWriter, ishttps bool) {
        secureheaders.SetHeaders(secureheaders.DefaultHeaders(), hw, ishttps)
        hw.Header().Set("Cache-Control", "no-store")
}

func (srv *Server) writeResult(hw http.ResponseWriter, statusCode int, err error) {
        SetHeaders(hw, srv.ishttps)
        hw.WriteHeader(statusCode)
        if err != nil {
                const tmpl = `<html><body><h2>%03d %s</h2><p>%s</p></body></html>`
                body := fmt.Appendf(nil, tmpl, statusCode, http.StatusText(statusCode), html.EscapeString(err.Error()))
                _, _ = hw.Write(body) // #nosec G705 -- error text is HTML-escaped
        }
}

// mergeUserInfoClaims accepts fallback claims only for the verified subject.
// Email verification stays paired with the email from the same response.
func mergeUserInfoClaims(dst, src map[string]any) {
        subject, _ := dst["sub"].(string)
        if other, _ := src["sub"].(string); subject == "" || other != subject {
                return
        }
        _, hasEmail := dst["email"]
        if !hasEmail {
                if email, ok := src["email"]; ok {
                        dst["email"] = email
                        dst["email_verified"] = src["email_verified"]
                }
        } else if _, hasVerified := dst["email_verified"]; !hasVerified {
                email, _ := dst["email"].(string)
                other, _ := src["email"].(string)
                if email != "" && email == other {
                        dst["email_verified"] = src["email_verified"]
                }
        }
        for k, v := range src {
                if k != "email_verified" {
                        if _, ok := dst[k]; !ok {
                                dst[k] = v
                        }
                }
        }
}

func (srv *Server) extractEmail(claims map[string]any) (sessEmailValue any, verified bool) {
        for _, k := range []string{"email", "mail", "public_email"} {
                if s, ok := claims[k].(string); ok {
                        if s = strings.TrimSpace(s); s != "" {
                                if m, err := mail.ParseAddress(s); err == nil {
                                        return foldEmailCase(m.Address), k == "email" && extractEmailVerified(claims)
                                }
                        }
                }
        }
        if l := srv.Jaws.Logger; l != nil {
                l.Warn("jawsauth: no email found")
        }
        return
}

func extractEmailVerified(claims map[string]any) (verified bool) {
        if claims != nil {
                switch value := claims["email_verified"].(type) {
                case bool:
                        verified = value
                case string:
                        verified, _ = strconv.ParseBool(value)
                case float64:
                        verified = value != 0
                case json.Number:
                        if number, err := value.Float64(); err == nil {
                                verified = number != 0
                        }
                }
        }
        return
}

func (srv *Server) fetchUserInfo(ctx context.Context, userinfoURL string, tokenSource oauth2.TokenSource) (userinfo map[string]any, err error) {
        if userinfoURL != "" && tokenSource != nil {
                client := oauth2.NewClient(ctx, tokenSource)
                client.CheckRedirect = func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }
                timeout := boundedProviderClient(client).Timeout
                requestCtx, cancel := context.WithTimeout(ctx, timeout)
                defer cancel()
                // oauth2.Transport logs from its deprecated CancelRequest; use the request context.
                client.Timeout = 0
                var request *http.Request
                if request, err = http.NewRequestWithContext(requestCtx, http.MethodGet, userinfoURL, nil); err != nil {
                        return nil, err
                }
                var resp *http.Response
                if resp, err = client.Do(request); /*#nosec G704*/ err == nil {
                        defer func() {
                                if closeErr := resp.Body.Close(); err == nil && closeErr != nil {
                                        err = closeErr
                                }
                        }()
                        if resp.StatusCode != http.StatusOK {
                                return nil, fmt.Errorf("%w %s", ErrUserInfoStatus, resp.Status)
                        }
                        var body []byte
                        if body, err = io.ReadAll(io.LimitReader(resp.Body, maxUserInfoBytes+1)); err == nil {
                                if len(body) > maxUserInfoBytes {
                                        err = ErrUserInfoTooLarge
                                } else if !json.Valid(body) {
                                        err = ErrUserInfoInvalidResponse
                                } else {
                                        decoder := json.NewDecoder(bytes.NewReader(body))
                                        if srv.profileSubject != nil {
                                                decoder.UseNumber()
                                        }
                                        err = decoder.Decode(&userinfo)
                                        if err != nil || userinfo == nil {
                                                userinfo, err = nil, ErrUserInfoInvalidResponse
                                        }
                                }
                        }
                }
        }
        return
}

func (srv *Server) addUserInfoClaims(ctx context.Context, claims map[string]any, tokenSource oauth2.TokenSource) {
        if fallback, err := srv.fetchUserInfo(ctx, srv.userinfoUrl, tokenSource); srv.Jaws.Log(err) == nil {
                mergeUserInfoClaims(claims, fallback)
        }
}

func (srv *Server) verifyOAuthIdentity(ctx context.Context, token *oauth2.Token, nonce string) (claims map[string]any, tokenSource oauth2.TokenSource, expiry time.Time, status int, err error) {
        status = http.StatusUnauthorized
        if srv.idTokenVerifier != nil {
                var raw string
                if token != nil {
                        raw, _ = token.Extra("id_token").(string)
                }
                idToken, verifiedClaims, verifyErr := srv.verifyIDToken(ctx, raw)
                if verifyErr != nil {
                        return nil, nil, time.Time{}, status, verifyErr
                }
                if nonce == "" {
                        return nil, nil, time.Time{}, status, ErrOIDCMissingNonce
                }
                if idToken.Nonce != nonce {
                        return nil, nil, time.Time{}, status, ErrOIDCNonceMismatch
                }
                claims = verifiedClaims
                tokenSource = srv.oauth2cfg.TokenSource(srv.oauth2Context(context.Background()), token)
                srv.addUserInfoClaims(ctx, claims, oauth2.StaticTokenSource(token))
                if err = srv.identityExpiryError(idToken.Expiry); err != nil {
                        return nil, nil, time.Time{}, status, err
                }
                return claims, tokenSource, idToken.Expiry, 0, nil
        }
        if srv.profileSubject == nil || !profileTokenUnexpired(token) {
                return nil, nil, time.Time{}, status, ErrOAuth2InvalidProfile
        }
        claims, err = srv.fetchUserInfo(ctx, srv.userinfoUrl, oauth2.StaticTokenSource(token))
        if err != nil {
                return nil, nil, time.Time{}, http.StatusBadGateway, err
        }
        // UserInfo can outlive the token; reject it before calling application code.
        if !profileTokenUnexpired(token) {
                return nil, nil, time.Time{}, http.StatusUnauthorized, ErrOAuth2InvalidProfile
        }
        var subject string
        subject, err = srv.profileSubject(claims)
        if srv.Jaws.Log(err) != nil || subject == "" {
                return nil, nil, time.Time{}, status, ErrOAuth2InvalidProfile
        }
        claims["iss"] = srv.issuer
        claims["sub"] = subject
        expiry = time.Now().Add(profileAuthLifetime)
        if !token.Expiry.IsZero() && token.Expiry.Before(expiry) {
                expiry = token.Expiry
        }
        if err = srv.identityExpiryError(expiry); err != nil {
                return nil, nil, time.Time{}, status, err
        }
        return claims, nil, expiry, 0, nil
}

func profileTokenUnexpired(token *oauth2.Token) bool {
        if token == nil || token.AccessToken == "" {
                return false
        }
        if token.Expiry.IsZero() {
                // oauth2 uses zero Expiry for both omitted and explicit zero expires_in.
                switch expiresIn := token.Extra("expires_in").(type) {
                case nil:
                        return true
                case string:
                        return expiresIn == "" // omitted in a form-encoded token response
                }
                return false
        }
        return token.Expiry.After(time.Now())
}

func (srv *Server) rotateOAuthSession(hw http.ResponseWriter, hr *http.Request, old *jaws.Session, flow pendingOAuthFlow) (rotated *jaws.Session, hadTimer, stale bool) {
        rotationLock := srv.SessionRotationLock
        if rotationLock != nil {
                rotationLock.Lock()
                defer rotationLock.Unlock()
        }
        if flow.revoked() || srv.Jaws.GetSession(hr) != old {
                return nil, false, true
        }
        if rotated = srv.Jaws.NewSession(hw, hr); rotated != nil {
                srv.mu.Lock()
                hadTimer = srv.authTimers[old.ID()] != nil
                srv.stopSessionAuthTimerLocked(old, nil)
                srv.mu.Unlock()
        }
        return
}

func (srv *Server) completeLink(sess *jaws.Session, hr *http.Request, claims map[string]any, expiry time.Time, flow pendingOAuthFlow) (status int, err error) {
        rotationLock := srv.SessionRotationLock
        if rotationLock != nil {
                rotationLock.Lock()
                defer rotationLock.Unlock()
        }
        if flow.revoked() || srv.Jaws.GetSession(hr) != sess {
                return http.StatusForbidden, ErrOAuth2MissingSession
        }
        if err = srv.identityExpiryError(expiry); err != nil {
                return http.StatusUnauthorized, err
        }
        status, err = srv.LinkEvent(sess, hr, claims)
        if srv.Jaws.Log(err) != nil || status != 0 {
                if status < 400 || status > 599 {
                        status = http.StatusForbidden
                }
                return status, ErrOAuth2LinkRejected
        }
        return http.StatusFound, nil
}

func (srv *Server) completeLogin(hw http.ResponseWriter, hr *http.Request, old *jaws.Session, flow pendingOAuthFlow, claims map[string]any, tokenSource oauth2.TokenSource, expiry time.Time) (sess *jaws.Session, status int, err error) {
        sess, status, err = old, http.StatusServiceUnavailable, ErrOAuth2MissingSession
        if !srv.RotateSessionOnLogin {
                status, err = srv.completeLoginInPlace(hr, old, flow, claims, tokenSource, expiry)
                return
        }
        rotated, hadTimer, stale := srv.rotateOAuthSession(hw, hr, old, flow)
        if rotated == nil {
                if stale {
                        status = http.StatusForbidden
                } else {
                        hw.Header().Set("Retry-After", "60")
                }
                return
        }
        if hadTimer && srv.LogoutEvent != nil {
                srv.LogoutEvent(old, hr)
        }
        sess = rotated
        err = srv.storeSessionAuthClaimsIfActive(sess, claims, tokenSource, expiry, nil, flow.canceled)
        if err != nil {
                http.SetCookie(hw, rotated.Close())
                if errors.Is(err, ErrOAuth2MissingSession) {
                        status = http.StatusForbidden
                } else {
                        status = http.StatusUnauthorized
                }
                return
        }
        return sess, http.StatusFound, nil
}

func (srv *Server) completeLoginInPlace(hr *http.Request, sess *jaws.Session, flow pendingOAuthFlow, claims map[string]any, tokenSource oauth2.TokenSource, expiry time.Time) (status int, err error) {
        if flow.revoked() {
                return http.StatusForbidden, ErrOAuth2MissingSession
        }
        // A logger can call Logout, which takes SessionRotationLock.
        email, verified := srv.extractEmail(claims)
        claims["email_verified"] = verified
        rotationLock := srv.SessionRotationLock
        if rotationLock != nil {
                rotationLock.Lock()
        }
        status, err = http.StatusForbidden, ErrOAuth2MissingSession
        var delay time.Duration
        var replaced bool
        if !flow.revoked() && srv.Jaws.GetSession(hr) == sess {
                delay, replaced, err = srv.storePreparedSessionAuthClaimsIfActive(sess, claims, tokenSource, expiry, nil, flow.canceled, email, verified)
                if err == nil {
                        srv.mu.Lock()
                        if srv.oauthCallbacks[sess].canceled == flow.canceled {
                                delete(srv.oauthCallbacks, sess)
                        }
                        srv.mu.Unlock()
                        status = http.StatusFound
                } else if !errors.Is(err, ErrOAuth2MissingSession) {
                        status = http.StatusUnauthorized
                }
        }
        if rotationLock != nil {
                rotationLock.Unlock()
        }
        if err == nil {
                srv.notifyStoredSessionAuthClaims(sess, expiry, delay, replaced)
        }
        return
}

func (srv *Server) completeOAuthCallback(ctx context.Context, hw http.ResponseWriter, hr *http.Request, sess *jaws.Session, flow pendingOAuthFlow) (*jaws.Session, int, error) {
        status, err := oauth2CallbackError(http.StatusBadRequest, hr)
        if err != nil {
                return sess, status, err
        }
        if flow.verifier == "" {
                return sess, http.StatusBadRequest, ErrOAuth2MissingPKCEVerifier
        }
        token, err := srv.oauth2cfg.Exchange(ctx, hr.FormValue("code") /* #nosec G120 */, oauth2.VerifierOption(flow.verifier))
        if srv.Jaws.Log(err) != nil {
                status := http.StatusBadGateway
                var retrieveErr *oauth2.RetrieveError
                if errors.As(err, &retrieveErr) && retrieveErr.Response != nil &&
                        retrieveErr.Response.StatusCode == http.StatusBadRequest {
                        status = http.StatusBadRequest
                }
                return sess, status, err
        }
        claims, tokenSource, expiry, status, err := srv.verifyOAuthIdentity(ctx, token, flow.nonce)
        if err != nil {
                return sess, status, err
        }
        if flow.link {
                status, err = srv.completeLink(sess, hr, claims, expiry, flow)
        } else {
                sess, status, err = srv.completeLogin(hw, hr, sess, flow, claims, tokenSource, expiry)
        }
        return sess, status, err
}

// HandleAuthResponse handles the OIDC or OAuth2 profile callback endpoint.
//
// For GET requests it validates the state, exchanges the authorization code using the
// stored PKCE verifier, verifies the ID token or authenticated profile, and by
// default stores claims in the existing session. A link flow invokes [Server.LinkEvent].
// It invokes [Server.LoginEvent] after a new sign-in or [Server.LoginFailed]
// on login or link failure. Non-GET requests receive 405.
//
// With [Server.RotateSessionOnLogin], login creates a new session and discards
// pre-login application data. If no session slot is available, existing
// authentication is preserved and the default response is 503. The user must
// restart login after capacity becomes available.
func (srv *Server) HandleAuthResponse(hw http.ResponseWriter, hr *http.Request) {
        if hr.Method != http.MethodGet {
                hw.Header().Set("Allow", http.MethodGet)
                srv.writeResult(hw, http.StatusMethodNotAllowed, ErrOAuth2Callback)
                return
        }
        providerCtx, cancel := context.WithTimeout(hr.Context(), oauth2ProviderTimeout)
        defer cancel()
        authctx := srv.oauth2Context(providerCtx)
        sess := srv.Jaws.GetSession(hr)
        var sessEmail string
        if sess != nil {
                sessEmail, _ = sess.Get(srv.SessionEmailKey).(string)
        }
        statusCode, err := http.StatusInternalServerError, ErrOAuth2NotConfigured
        var flow pendingOAuthFlow
        if srv.oauth2cfg != nil {
                statusCode, err = http.StatusBadRequest, ErrOAuth2MissingSession
                if sess != nil {
                        flow, err = srv.takeOAuthFlow(sess, hr.FormValue("state") /* #nosec G120 */)
                        if err == nil {
                                defer srv.finishOAuthFlow(sess, flow)
                                sess, statusCode, err = srv.completeOAuthCallback(authctx, hw, hr, sess, flow)
                                if err == nil && !flow.link {
                                        sessEmail, _ = sess.Get(srv.SessionEmailKey).(string)
                                }
                        }
                }
        }
        if err == nil && !flow.link {
                srv.emitLoginEvent(sess, hr)
                current, _ := srv.sessionAuthStatus(sess, time.Now)
                if flow.revoked() || !current || !sessionAlive(sess) {
                        if srv.RotateSessionOnLogin {
                                srv.clearSessionAuth(sess, hr, nil, false)
                                http.SetCookie(hw, sess.Close())
                        }
                        err = ErrOAuth2MissingSession
                        statusCode = http.StatusForbidden
                } else if !srv.RotateSessionOnLogin {
                        reloadAuthRequests(sess.Requests())
                }
        }
        if err == nil {
                hw.Header().Set("Location", flow.referrer)
        }
        if err != nil && srv.LoginFailed != nil {
                if srv.LoginFailed(hw, hr, statusCode, err, sessEmail) {
                        return
                }
        }
        srv.writeResult(hw, statusCode, err)
}

// ErrServerNilJaws is returned by New and NewDebug when the *jaws.Jaws
// argument is nil. Server methods dereference Server.Jaws and require it
// to be non-nil for the lifetime of the Server.
var ErrServerNilJaws = errors.New("jawsauth: nil *jaws.Jaws")

func foldEmailCase(s string) string {
        return strings.Map(func(r rune) rune {
                if r >= 'A' && r <= 'Z' {
                        return r + ('a' - 'A')
                }
                return r
        }, s)
}

func normalizeEmail(s string) (email string) {
        s = strings.TrimSpace(s)
        if m, e := mail.ParseAddress(s); e == nil {
                s = m.Address
        }
        email = foldEmailCase(s)
        return
}

func callbackPathFromURL(u *url.URL) (callbackPath string) {
        rawPath := u.Path
        callbackPath = path.Clean(rawPath)
        if callbackPath == "" || callbackPath == "." {
                callbackPath = "/"
        } else if strings.HasSuffix(rawPath, "/") && callbackPath != "/" {
                callbackPath += "/"
        }
        return
}

// HandleFunc registers handler to serve requests for the given URI path.
//
// It matches the shape of http.ServeMux.Handle and is supplied to New and NewDebug
// so the login, logout and callback endpoints can be wired into the caller's router.
type HandleFunc func(uri string, handler http.Handler)

// EventFunc is called for login and logout lifecycle events.
//
// jawsauth calls [Server.LogoutEvent] after releasing SessionRotationLock.
// When LoginEvent is set, clearing auth before it begins suppresses both
// events; clearing auth during it defers LogoutEvent until it returns.
// LoginEvent can read claims from sess with [Server.SessionKey]. During in-place
// login, [Server.CurrentClaims] returns nil until the event ends.
// LogoutEvent receives a nil request for timer cleanup, deferred delivery, or
// a call to [Server.Logout] with a nil request.
// With [Server.RotateSessionOnLogin], re-login calls LogoutEvent on the old
// session before storing the new sign-in; calling Logout there cancels it.
// For cross-provider cleanup, check the other Server's CurrentClaims before
// calling Logout; that call can revoke a running sign-in.
type EventFunc func(sess *jaws.Session, hr *http.Request)

// FailedFunc handles OAuth callback failures and flow conflicts.
//
// It receives login and link start conflicts, and protected GETs while an
// in-place login notification is pending. Other flow-start failures write their
// own response.
//
// It returns true if it wrote the HTTP response itself, in which case jawsauth writes
// nothing further. Errors can contain untrusted browser-supplied text. Quote errors
// when logging them and escape them for the output format when rendering them.
type FailedFunc func(hw http.ResponseWriter, hr *http.Request, httpCode int, err error, email string) (wroteresponse bool)

// LinkFunc receives the current session and a newly authenticated identity. Recheck
// the existing authentication before a durable link write; with multiple
// providers, use the original provider's CurrentClaims. Logout and flow
// cancellation affect only their own Server. jawsauth holds
// SessionRotationLock while calling LinkFunc; do not acquire it again or call
// Server.Logout in the callback. Claims contain raw provider fields, so their
// types (including email_verified) may differ from stored session values.
// Return zero and nil on success, or an HTTP error status.
type LinkFunc func(sess *jaws.Session, hr *http.Request, claims map[string]any) (statusCode int, err error)

// Server provides OIDC or OAuth2 profile authentication for JaWS sessions.
//
// Create one with New or NewDebug. Its methods are safe for concurrent use; the
// exported configuration fields and callbacks should be set before serving requests.
type Server struct {
        Jaws *jaws.Jaws
        // New appends Config.Namespace to these four session keys when configured.
        SessionKey              string              // default is "oidc_claims", value will be of type map[string]any
        SessionTokenKey         string              // default is "oauth2_tokensource"; an oauth2.TokenSource in OIDC mode, unset in profile mode
        SessionEmailKey         string              // default is "email", value is a parsed string address with ASCII letters lowercased
        SessionEmailVerifiedKey string              // default is "email_verified", value will be of type bool
        HandledPaths            map[string]struct{} // URI paths excluded from OAuth return targets
        LoginEvent              EventFunc           // called after claims are stored; revocation may suppress or follow it
        LogoutEvent             EventFunc           // auth-clear callback; see EventFunc for ordering and nil requests
        LoginFailed             FailedFunc          // callback failures and flow conflicts; see FailedFunc
        LinkEvent               LinkFunc            // required by HandleLink; register its POST route separately
        // SessionRotationLock is held during flow start, rotation, in-place login
        // claim storage, auth clearing, and LinkEvent.
        // New supplies a shared package lock; set the same custom lock on servers using
        // one JaWS instance when coordinating with other authentication methods.
        SessionRotationLock sync.Locker
        Options             []oauth2.AuthCodeOption // authorization URL options, e.g. oauth2.AccessTypeOffline; never sent to token exchange
        // RotateSessionOnLogin replaces the JaWS Session during login.
        //
        // It defaults to false. When enabled, login publishes a new cookie, closes
        // the old Session, and requires a free slot under JaWS session limits.
        // Set it before serving requests.
        RotateSessionOnLogin bool

        // RequireVerifiedAdminEmail requires email_verified for a non-empty admin list.
        //
        // It defaults to true in profile mode and false in OIDC mode. When true,
        // providers that omit email_verified cannot grant admin access through a
        // non-empty list. Set before serving requests.
        // Ordinary login is unaffected.
        RequireVerifiedAdminEmail bool

        oauth2cfg          *oauth2.Config
        idTokenVerifier    *oidc.IDTokenVerifier
        userinfoUrl        string
        profileSubject     func(map[string]any) (string, error)
        issuer             string
        httpClient         *http.Client
        ishttps            bool
        mu                 sync.Mutex          // protects following fields and session email/verification pairs
        admins             map[string]struct{} // if not empty, emails of admins
        handle403          http.Handler        // handler for 403 Forbidden
        authTimers         map[uint64]*authTimerState
        loginEvents        map[*jaws.Session]*pendingLoginEvent
        oauthCallbacks     map[*jaws.Session]pendingOAuthFlow
        authTimerAfterFunc authTimerAfterFunc
        namespace          string
}

var defaultSessionRotationLock sync.Mutex

// NewDebug behaves like New but can override the scheme and host of cfg.RedirectURL.
//
// overrideUrl supplies the replacement scheme and host (an empty overrideUrl disables
// the override), which is useful when serving behind a different public address during
// development.
//
// A nil jw returns ErrServerNilJaws and a nil Server. Otherwise a non-nil Server is
// always returned. Authentication is configured, and the login, logout and callback
// handlers registered via handleFn, only when cfg, handleFn and cfg.RedirectURL are
// all provided; setup errors are returned with the Server.
func NewDebug(jw *jaws.Jaws, cfg *Config, handleFn HandleFunc, overrideUrl string) (srv *Server, err error) {
        if jw == nil {
                err = ErrServerNilJaws
                return
        }
        srv = &Server{
                Jaws:                    jw,
                SessionKey:              "oidc_claims",
                SessionTokenKey:         "oauth2_tokensource",
                SessionEmailKey:         "email",
                SessionEmailVerifiedKey: "email_verified",
                HandledPaths:            make(map[string]struct{}),
                SessionRotationLock:     &defaultSessionRotationLock,
                admins:                  make(map[string]struct{}),
                handle403:               default403handler{},
                authTimers:              make(map[uint64]*authTimerState),
                authTimerAfterFunc:      realAuthTimerAfterFunc,
        } // #nosec G101
        if cfg != nil {
                srv.namespace = cfg.Namespace
                srv.issuer = cfg.Issuer
                srv.SessionKey = srv.sessionKey(srv.SessionKey)
                srv.SessionTokenKey = srv.sessionKey(srv.SessionTokenKey)
                srv.SessionEmailKey = srv.sessionKey(srv.SessionEmailKey)
                srv.SessionEmailVerifiedKey = srv.sessionKey(srv.SessionEmailVerifiedKey)
        }
        if cfg != nil && handleFn != nil && cfg.RedirectURL != "" {
                setupCtx, cancel := context.WithTimeout(context.Background(), oauth2ProviderTimeout)
                defer cancel()
                if srv.oauth2cfg, srv.userinfoUrl, srv.idTokenVerifier, err = cfg.buildContext(setupCtx, overrideUrl); err == nil {
                        srv.httpClient = cfg.HTTPClient
                        srv.profileSubject = cfg.ProfileSubject
                        srv.RequireVerifiedAdminEmail = srv.profileSubject != nil
                        var u *url.URL
                        if u, err = url.Parse(srv.oauth2cfg.RedirectURL); err == nil {
                                srv.ishttps = (u.Scheme == "https")
                                callbackPath := callbackPathFromURL(u)
                                dir := path.Dir(path.Clean(callbackPath))
                                srv.handlePath(callbackPath, handleFn, http.HandlerFunc(srv.HandleAuthResponse))
                                srv.handlePath(path.Join(dir, "login"), handleFn, http.HandlerFunc(srv.HandleLogin))
                                srv.handlePath(path.Join(dir, "logout"), handleFn, http.HandlerFunc(srv.HandleLogout))
                                jw.MakeAuth = srv.makeAuth
                        }
                }
        }
        return
}

// New creates a Server providing OIDC or OAuth2 profile authentication for JaWS sessions.
//
// It configures the Server from cfg and registers the login, logout and OAuth2
// callback endpoints via handleFn. A nil jw returns ErrServerNilJaws. Check both
// err and Valid before serving protected routes; an invalid Server leaves its
// wrapped handlers unprotected.
func New(jw *jaws.Jaws, cfg *Config, handleFn HandleFunc) (srv *Server, err error) {
        return NewDebug(jw, cfg, handleFn, "")
}

func (srv *Server) makeAuth(rq *jaws.Request) jaws.Auth {
        return &JawsAuth{server: srv, sess: srv.Jaws.GetSession(rq.Initial())}
}

func (srv *Server) sessionKey(key string) string {
        if srv.namespace != "" {
                return key + "." + srv.namespace
        }
        return key
}

func (srv *Server) handlePath(p string, handleFn HandleFunc, h http.Handler) {
        if _, ok := srv.HandledPaths[p]; !ok {
                srv.HandledPaths[p] = struct{}{}
                handleFn(p, h)
        }
}

// IsAdmin returns true if email belongs to an admin, if the list of admins is empty, or if srv is nil.
//
// It parses a raw address using the rules in [Server.SetAdmins]. This lookup cannot
// verify email ownership. [Server.WrapAdmin], [Server.HandlerAdmin] and
// [JawsAuth.IsAdmin] also enforce [Server.RequireVerifiedAdminEmail].
func (srv *Server) IsAdmin(email string) (yes bool) {
        yes = true
        if srv != nil {
                email = normalizeEmail(email)
                srv.mu.Lock()
                _, yes = srv.admins[email]
                yes = yes || len(srv.admins) == 0
                srv.mu.Unlock()
        }
        return
}

func (srv *Server) sessionIsAdmin(sess *jaws.Session) bool {
        if srv == nil {
                return true
        }
        srv.mu.Lock()
        defer srv.mu.Unlock()
        return srv.sessionIsAdminLocked(sess, srv.admins)
}

// Caller holds srv.mu.
func (srv *Server) sessionIsAdminLocked(sess *jaws.Session, admins map[string]struct{}) bool {
        if len(admins) == 0 {
                return true
        }
        if sess == nil {
                return false
        }
        email, _ := sess.Get(srv.SessionEmailKey).(string)
        verified, _ := sess.Get(srv.SessionEmailVerifiedKey).(bool)
        // Session emails are already parsed; parsing again can change quoted local parts.
        _, listed := admins[foldEmailCase(email)]
        return listed && (!srv.RequireVerifiedAdminEmail || verified)
}

// SetAdmins sets the emails of administrators.
//
// Valid addresses are parsed with [mail.ParseAddress]; other nonempty entries
// are retained. ASCII letters are lowercased.
// Non-ASCII characters are matched exactly. If empty, everyone is considered an
// administrator. Session checks also apply [Server.RequireVerifiedAdminEmail].
//
// Sessions losing admin access have their JaWS requests retired with a page reload;
// they remain authenticated and can reopen non-admin pages.
func (srv *Server) SetAdmins(emails []string) {
        if srv == nil {
                return
        }
        admins := make(map[string]struct{})
        for _, email := range emails {
                if email = normalizeEmail(email); email != "" {
                        admins[email] = struct{}{}
                }
        }
        var sessions []*jaws.Session
        if srv.Jaws != nil {
                sessions = srv.Jaws.Sessions()
        }
        var requests []*jaws.Request
        srv.mu.Lock()
        for _, sess := range sessions {
                if sess.Get(srv.SessionKey) != nil && srv.sessionIsAdminLocked(sess, srv.admins) && !srv.sessionIsAdminLocked(sess, admins) {
                        requests = append(requests, sess.Requests()...)
                }
        }
        srv.admins = admins
        srv.mu.Unlock()
        reloadAuthRequests(requests)
}

// GetAdmins returns a sorted list of the administrator emails. If empty, everyone is considered an administrator.
func (srv *Server) GetAdmins() (emails []string) {
        if srv != nil {
                srv.mu.Lock()
                for k := range srv.admins {
                        emails = append(emails, k)
                }
                srv.mu.Unlock()
                sort.Strings(emails)
        }
        return
}

// Set403Handler sets the handler used to serve 403 Forbidden responses to
// authenticated non-admin users.
//
// A nil h restores the default handler. It is safe for concurrent use.
func (srv *Server) Set403Handler(h http.Handler) {
        if h == nil {
                h = default403handler{}
        }
        srv.mu.Lock()
        srv.handle403 = h
        srv.mu.Unlock()
}

func (srv *Server) get403Handler() (h http.Handler) {
        h = default403handler{}
        if srv != nil {
                srv.mu.Lock()
                if srv.handle403 != nil {
                        h = srv.handle403
                }
                srv.mu.Unlock()
        }
        return
}

// Valid returns true if OIDC or OAuth2 profile authentication is configured.
func (srv *Server) Valid() bool {
        return srv != nil && srv.oauth2cfg != nil && (srv.idTokenVerifier != nil || srv.profileSubject != nil)
}

// wrap returns a http.Handler that requires an authenticated user before invoking h.
//
// When admin is true, non-admins receive 403. Unauthenticated GET begins login;
// other unauthenticated methods receive 403. An invalid Server returns h.
func (srv *Server) wrap(h http.Handler, admin bool) (rh http.Handler) {
        rh = h
        if srv.Valid() {
                rh = wrapper{server: srv, handler: h, admin: admin}
        }
        return
}

// WrapAdmin returns a http.Handler that requires an authenticated administrator
// before invoking h.
//
// Unauthenticated GET begins login; other unauthenticated methods receive 403.
// Authenticated non-admins receive 403. [Server.RequireVerifiedAdminEmail]
// applies. A GET request during a running login callback may return 409.
// An invalid Server returns h.
func (srv *Server) WrapAdmin(h http.Handler) (rh http.Handler) {
        return srv.wrap(h, true)
}

// Wrap returns a http.Handler that requires an authenticated user before invoking h.
//
// Unauthenticated GET begins login; other unauthenticated methods receive 403.
// A GET request during a running login callback may return 409.
// An invalid Server returns h.
func (srv *Server) Wrap(h http.Handler) (rh http.Handler) {
        return srv.wrap(h, false)
}

// HandlerAdmin returns a http.Handler that renders the named jaws.Template with dot
// and requires an authenticated administrator.
//
// Unauthenticated GET begins login; other unauthenticated methods receive 403.
// Authenticated non-admins receive 403. [Server.RequireVerifiedAdminEmail]
// applies. A GET request during a running login callback may return 409.
// An invalid, non-nil Server returns the unprotected template handler.
func (srv *Server) HandlerAdmin(name string, dot any) http.Handler {
        return srv.wrap(ui.Handler(srv.Jaws, name, dot), true)
}

func (srv *Server) Handler(name string, dot any) http.Handler {
        return srv.wrap(ui.Handler(srv.Jaws, name, dot), false)
}

package jawsauth

import "net/http"

type wrapper struct {
        server  *Server
        handler http.Handler
        admin   bool
}

func (w wrapper) ServeHTTP(hw http.ResponseWriter, hr *http.Request) {
        sess := w.server.Jaws.GetSession(hr)
        current, present, pending := w.server.sessionAccessStatus(sess)
        if pending {
                if hr.Method == http.MethodGet {
                        if w.server.LoginFailed != nil {
                                email, _ := sess.Get(w.server.SessionEmailKey).(string)
                                if w.server.LoginFailed(hw, hr, http.StatusConflict, ErrOAuth2LoginInProgress, email) {
                                        return
                                }
                        }
                        w.server.writeResult(hw, http.StatusConflict, ErrOAuth2LoginInProgress)
                } else {
                        w.server.writeResult(hw, http.StatusForbidden, nil)
                }
                return
        }
        if !current {
                if present {
                        w.server.clearSessionAuth(sess, hr, nil, true)
                }
                if hr.Method != http.MethodGet {
                        w.server.writeResult(hw, http.StatusForbidden, nil)
                        return
                }
                w.server.startOAuth(hw, hr, false, true)
                return
        }

        if w.admin && !w.server.sessionIsAdmin(sess) {
                w.server.get403Handler().ServeHTTP(hw, hr)
                return
        }
        w.handler.ServeHTTP(hw, hr)
        // A revocation may take its request snapshot before rendering attaches a
        // JaWS request. Retire that request before this response completes.
        if current, _, pending := w.server.sessionAccessStatus(sess); !current || pending || (w.admin && !w.server.sessionIsAdmin(sess)) {
                reloadAuthRequests(sess.Requests())
        }
}
